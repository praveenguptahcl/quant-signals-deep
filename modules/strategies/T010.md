        tickets = [ticket(side='SELL', symbol='VARIANCE', qty=size(...), limit=touch(t), tif='DAY',
                          parent_signal=f"S030@{t.bar_open_ts}")]

    for tk in tickets: log_decision(tk, inputs_hash, params_hash)  # C5 / App. g
