```text
function emit(state, signals, bars, cfg) -> list[OrderTicket]:
    assert quotes non-empty else return UNKNOWN                     # F1
    q = signals['S003']   # newest L1 snapshot with imbalance
    vpin = signals['S008'].vpin
    assert isfinite(q.imbalance) and isfinite(vpin) else return UNKNOWN  # F2
    if vpin > cfg.vpin_cancel: cancel_quotes(); return []          # C12
    if abs(state.inventory_q) > cfg.inv_max: return []            # C11

    edge_bps = 4.0 - vpin * 10.0                                   # [example]
    cost_ok = expected_cost_bps(notional, adv_pct, venue, side, urgency) <= cfg.cost_gate_k * edge_bps
    # ^^^ normative cost-gate predicate: expected_cost_bps(...) <= k * edge_bps

    tickets = []
    if abs(q.imbalance) >= cfg.I_quote and cost_ok:
        side = 'BUY' if q.imbalance > 0 else 'SELL'   # heavy side
        px = q.bid if side == 'BUY' else q.ask
        tickets = [ticket(side=side, qty=100, limit=px, tif='DAY', maker=True,
                          parent_signal=f"S003@{q.event_ts}")]
    for tk in tickets: log_decision(tk, inputs_hash, params_hash)  # C5 / App. g
    assert fill.event_ts > q.event_ts for any fill                 # t->t+1 causality
    return tickets
```
